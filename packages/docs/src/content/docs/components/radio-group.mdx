Radio groups are used to group multiple [radios](/docs/components/radio) so they function as a single form control.

```demo:open
<pc-radio-group label="Select an option" name="group-0" value="option-1">
    <pc-radio value="option-1">Option 1</pc-radio>
    <pc-radio value="option-2">Option 2</pc-radio>
    <pc-radio value="option-3">Option 3</pc-radio>
</pc-radio-group>
```

## Demos

### Checked

Use the `value` attribute on the radio group to set the checked radio.

```demo
<pc-radio-group label="Select an option" name="group-1" value="option-2">
    <pc-radio value="option-1">Option 1</pc-radio>
    <pc-radio value="option-2">Option 2</pc-radio>
    <pc-radio value="option-3">Option 3</pc-radio>
</pc-radio-group>
```

### Hints

Add a descriptive hint to a radio group with the `hint` attribute. For hints that contain HTML, use the `hint` slot instead.

```demo
<pc-radio-group
    label="Select an option"
    hint="Select your favourite option."
    name="group-2"
    value="option-1"
>
    <pc-radio value="option-1">Option 1</pc-radio>
    <pc-radio value="option-2">Option 2</pc-radio>
    <pc-radio value="option-3">Option 3</pc-radio>
</pc-radio-group>
```

### Segmented radios

Set the `appearance` attribute to `segmented` on all radios to turn them into segmented radios.

```demo
<div class="pc-stack">
    <pc-radio-group
        label="Horizontal options"
        hint="Select an option that makes you proud."
        orientation="horizontal"
        name="group-3"
        value="option-1"
    >
        <pc-radio appearance="segmented" value="option-1">Option 1</pc-radio>
        <pc-radio appearance="segmented" value="option-2">Option 2</pc-radio>
        <pc-radio appearance="segmented" value="option-3">Option 3</pc-radio>
    </pc-radio-group>

    <pc-radio-group
        label="Vertical options"
        hint="Select an option that makes you proud."
        orientation="vertical"
        name="group-4"
        value="option-1"
        style="max-inline-size: 300px"
    >
        <pc-radio appearance="segmented" value="option-1">Option 1</pc-radio>
        <pc-radio appearance="segmented" value="option-2">Option 2</pc-radio>
        <pc-radio appearance="segmented" value="option-3">Option 3</pc-radio>
    </pc-radio-group>
</div>
```

### Disabled

To disable the entire radio group, add the `disabled` attribute to the radio group.

```demo
<pc-radio-group label="Select an option" name="group-5" disabled>
    <pc-radio value="option-1">Option 1</pc-radio>
    <pc-radio value="option-2">Option 2</pc-radio>
    <pc-radio value="option-3">Option 3</pc-radio>
</pc-radio-group>
```

To disable individual radios, add the `disabled` attribute to the respective options.

```demo
<pc-radio-group label="Select an option" name="group-6">
    <pc-radio value="option-1">Option 1</pc-radio>
    <pc-radio value="option-2" disabled>Option 2</pc-radio>
    <pc-radio value="option-3">Option 3</pc-radio>
</pc-radio-group>
```

### Size

The size of [radios](/docs/components/radio) will be determined by the radio group’s `size` attribute.

```demo
<pc-radio-group
    class="radio-group-size"
    size="medium"
    label="Select an option"
    value="medium"
>
    <pc-radio value="small">Small</pc-radio>
    <pc-radio value="medium">Medium</pc-radio>
    <pc-radio value="large">Large</pc-radio>
</pc-radio-group>

<script>
    const radioGroup = document.querySelector(".radio-group-size");

    radioGroup.addEventListener("change", () => {
        radioGroup.size = radioGroup.value;
    });
</script>
```

If you need to have radios of different sizes, add the `size` attribute on the individual radios instead.

```demo
<pc-radio-group label="Select a mixed‐size option" value="medium">
    <pc-radio size="small" value="small">Small</pc-radio>
    <pc-radio size="medium" value="medium">Medium</pc-radio>
    <pc-radio size="large" value="large">Large</pc-radio>
</pc-radio-group>
```

### Horizontal

Set the `orientation` attribute to `horizontal` to make a horizontal radio group.

```demo
<pc-radio-group
    label="Select an option"
    name="group-7"
    value="option-1"
    orientation="horizontal"
>
    <pc-radio value="option-1">Option 1</pc-radio>
    <pc-radio value="option-2" disabled>Option 2</pc-radio>
    <pc-radio value="option-3">Option 3</pc-radio>
</pc-radio-group>
```

### Required

Set the `required` attribute to make selecting an option mandatory. If a value hasn’t been selected, it will prevent the form from submitting and display an error message.

```demo
<form class="radio-group-required pc-stack">
    <pc-radio-group label="Select an option" name="group-8" required>
        <pc-radio value="option-1">Option 1</pc-radio>
        <pc-radio value="option-2">Option 2</pc-radio>
        <pc-radio value="option-3">Option 3</pc-radio>
    </pc-radio-group>

    <div class="buttons">
        <pc-button appearance="primary" type="submit">Submit</pc-button>
        <pc-button appearance="danger" variant="plain" type="reset">Reset</pc-button>
    </div>
</form>

<script>
    const form = document.querySelector(".radio-group-required");

    form.addEventListener("submit", (event) => {
        event.preventDefault();

        alert("All fields are valid!");
    });
</script>
```

### Custom validity

Use the `setCustomValidity()` method to set a custom validation message. This will prevent the form from submitting and make the browser display the error message you provide. If you want to restore the default validation message, pass an empty string to the `setCustomValidity()` method.

```demo
<form class="radio-group-custom-validity pc-stack">
    <pc-radio-group label="Select an option" name="group-9" value="option-1">
        <pc-radio value="not-me">Not me</pc-radio>
        <pc-radio value="me-neither">Me neither</pc-radio>
        <pc-radio value="select-me">Select me</pc-radio>
    </pc-radio-group>

    <div class="buttons">
        <pc-button appearance="primary" type="submit">Submit</pc-button>
    </div>
</form>

<script>
    const form = document.querySelector(".radio-group-custom-validity");
    const radioGroup = form.querySelector("pc-radio-group");
    const errorMessage = "You must select the last option.";

    customElements.whenDefined("pc-radio").then(() => {
        radioGroup.setCustomValidity(errorMessage);
    });

    form.addEventListener("change", () => {
        const isValid = radioGroup.value === "select-me";
        radioGroup.setCustomValidity(isValid ? "" : errorMessage);
    });

    form.addEventListener("submit", (event) => {
        event.preventDefault();
        alert("All fields are valid!");
    });
</script>
```
